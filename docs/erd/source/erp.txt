orders
----
order_id PK int
customer_id int FK >- customers.customer_id
order_date date
status string

order_items
----
order_item_id PK int
order_id int FK >- orders.order_id
product_name string
quantity int
unit_price decimal(10,2)

payments
----
payment_id PK int
order_id int FK >- orders.order_id
payment_method string
amount decimal(10,2)
